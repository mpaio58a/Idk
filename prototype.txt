    local function DoPhysicsFling(TargetPlayer)
        if not TargetPlayer or not TargetPlayer.Character then return end

        local TargetRoot = TargetPlayer.Character:FindFirstChild("HumanoidRootPart")
        local TargetHum = TargetPlayer.Character:FindFirstChildOfClass("Humanoid")
        local Character = LocalPlayer.Character
        local Root = Character and Character:FindFirstChild("HumanoidRootPart")
        local Hum = Character and Character:FindFirstChildOfClass("Humanoid")

        if not TargetRoot or not TargetHum or not Root or not Hum then return end

        local OldPos = Root.CFrame
        local OldFPDH = workspace.FallenPartsDestroyHeight
        workspace.FallenPartsDestroyHeight = -50000
        workspace.CurrentCamera.CameraSubject = TargetHum

        Hum.PlatformStand = true
        local BV = Instance.new("BodyVelocity")
        BV.Parent = Root
        BV.MaxForce = Vector3.new(math.huge, math.huge, math.huge)
        BV.Velocity = Vector3.new(0, 0, 0)

        local StartTime = tick()
        local Connection
        local HasFlung = false

        Connection = RunService.Heartbeat:Connect(function()
            if tick() - StartTime > 3 or not TargetPlayer.Character or not Character.Parent or HasFlung then
                if Connection then Connection:Disconnect() end
                return
            end
            if TargetRoot.AssemblyLinearVelocity.Magnitude > 250 then
                HasFlung = true
                return
            end
            for _, part in pairs(Character:GetDescendants()) do
                if part:IsA("BasePart") then part.CanCollide = false end
            end
            local ChaosRot = CFrame.Angles(math.rad(math.random(-36000, 36000)), math.rad(math.random(-36000, 36000)), math.rad(math.random(-36000, 36000)))
            Root.CFrame = TargetRoot.CFrame * ChaosRot
            Root.AssemblyLinearVelocity = Vector3.new(9e30, 9e30, 9e30)
            Root.AssemblyAngularVelocity = Vector3.new(9e30, 9e30, 9e30)
        end)

        local elapsed = 0
        while elapsed < 3 do
            if HasFlung or not TargetPlayer.Character or not TargetPlayer.Character:FindFirstChild("HumanoidRootPart") then break end
            elapsed = elapsed + task.wait(0.1)
        end

        if Connection then Connection:Disconnect() end
        if BV then BV:Destroy() end
        Hum.PlatformStand = false
        workspace.CurrentCamera.CameraSubject = Hum
        workspace.FallenPartsDestroyHeight = OldFPDH
        Root.CFrame = OldPos
        Root.AssemblyLinearVelocity = Vector3.new(0, 0, 0)
        Root.AssemblyAngularVelocity = Vector3.new(0, 0, 0)
    end

local FolderCache = {}
local ActiveESPs = {}

local function GetTargetFolder(name)
    if FolderCache[name] and FolderCache[name].Parent then return FolderCache[name] end
    local f = workspace:FindFirstChild(name) or (workspace:FindFirstChild("Map") and workspace.Map:FindFirstChild(name))
    if not f then f = workspace:FindFirstChild(name, true) end
    FolderCache[name] = f
    return f
end

local function HandleTransparency(part, espName)
    if part:IsA("BasePart") then
        if part.Name == "HumanoidRootPart" then return end
        if part.Name == "Handle" and part.Transparency >= 1 then return end
        if part.Transparency >= 1 then
            local ancestorModel = part:FindFirstAncestorWhichIsA("Model")
            if ancestorModel and ancestorModel:FindFirstChildOfClass("Humanoid") then return end
        end

        if part:GetAttribute("OrigTrans_" .. espName) == nil then
            part:SetAttribute("OrigTrans_" .. espName, part.Transparency)
        end
        part.Transparency = 0 
    end
end

local function CreateESP(target, colorKey, espName, displayText)
    if not target or not target.Parent then return end

    pcall(function()
        if espName == "ESP_Puzzle" then
            if target:IsA("BasePart") then 
                HandleTransparency(target, espName)
            elseif target:IsA("Model") then
                for _, v in ipairs(target:GetDescendants()) do HandleTransparency(v, espName) end
            end
        end

        local root = target:IsA("Model") and (target.PrimaryPart or target:FindFirstChild("HumanoidRootPart") or target:FindFirstChild("Head") or target:FindFirstChildWhichIsA("BasePart", true)) or (target:IsA("BasePart") and target)

        if root then
            local currentColor = Options[colorKey].Value

            local hl = target:FindFirstChild(espName .. "_HL") or Instance.new("Highlight")
            hl.Name = espName .. "_HL"
            hl.FillColor = currentColor
            hl.FillTransparency = 0.5
            hl.OutlineColor = Color3.fromRGB(255, 255, 255)
            hl.DepthMode = Enum.HighlightDepthMode.AlwaysOnTop
            hl.Enabled = Toggles.HighlightEsp.Value
            hl.Parent = target

            local bg = root:FindFirstChild(espName .. "_BG") or Instance.new("BillboardGui")
            bg.Name = espName .. "_BG"
            bg.Size = UDim2.new(0, 200, 0, 40)
            bg.AlwaysOnTop = true
            bg.ExtentsOffset = (espName == "ESP_Player") and Vector3.new(0, 3.5, 0) or Vector3.new(0, 2, 0)

            local txt = bg:FindFirstChild("NameLabel") or Instance.new("TextLabel")
            txt.Name = "NameLabel"
            txt.Size = UDim2.new(1, 0, 1, 0)
            txt.BackgroundTransparency = 1
            txt.TextColor3 = currentColor
            txt.Font = Enum.Font.SourceSansBold
            txt.TextSize = 14
            txt.TextStrokeTransparency = 0
            txt.Parent = bg
            bg.Parent = root

            local boxBg = root:FindFirstChild(espName .. "_BoxBG") or Instance.new("BillboardGui")
            boxBg.Name = espName .. "_BoxBG"
            boxBg.Size = (espName == "ESP_Player") and UDim2.new(4, 0, 5.5, 0) or UDim2.new(2, 0, 2, 0)
            boxBg.ExtentsOffset = (espName == "ESP_Player") and Vector3.new(0, -0.2, 0) or Vector3.new(0, 0, 0)
            boxBg.AlwaysOnTop = true
            boxBg.Enabled = Toggles.BoxEsp.Value

            local boxFrame = boxBg:FindFirstChild("BoxFrame") or Instance.new("Frame")
            boxFrame.Name = "BoxFrame"
            boxFrame.Size = UDim2.new(1, 0, 1, 0)
            boxFrame.BackgroundTransparency = 1
            boxFrame.Parent = boxBg

            local stroke = boxFrame:FindFirstChild("UIStroke") or Instance.new("UIStroke")
            stroke.Color = currentColor
            stroke.Thickness = 1.5
            stroke.Parent = boxFrame
            boxBg.Parent = root

            ActiveESPs[target] = {
                Root = root,
                NameLabel = txt,
                Highlight = hl,
                Stroke = stroke,
                BoxBg = boxBg,
                BaseName = displayText or target.Name,
                ColorKey = colorKey
            }
        end
    end)
end

RunService.RenderStepped:Connect(function()
    if not LocalPlayer.Character or not LocalPlayer.Character:FindFirstChild("HumanoidRootPart") then return end
    local myPos = LocalPlayer.Character.HumanoidRootPart.Position

    for target, data in pairs(ActiveESPs) do
        if not target or not target.Parent or not data.Root or not data.Root.Parent then
            ActiveESPs[target] = nil
            continue
        end

        local dist = math.floor((myPos - data.Root.Position).Magnitude)
        local finalText = ""

        if Toggles.NameEsp.Value then finalText = data.BaseName end
        if Toggles.DistanceEsp.Value then
            finalText = finalText ~= "" and (finalText .. " [" .. tostring(dist) .. "m]") or ("[" .. tostring(dist) .. "m]")
        end

        data.NameLabel.Text = finalText
        data.NameLabel.Visible = (Toggles.NameEsp.Value or Toggles.DistanceEsp.Value)

        local currentColor = Options[data.ColorKey].Value
        if data.Highlight then 
            data.Highlight.FillColor = currentColor 
            data.Highlight.Enabled = Toggles.HighlightEsp.Value
        end
        if data.NameLabel then data.NameLabel.TextColor3 = currentColor end
        if data.Stroke then data.Stroke.Color = currentColor end
        if data.BoxBg then data.BoxBg.Enabled = Toggles.BoxEsp.Value end
    end
end)

local function RemoveESP(espName)
    pcall(function()
        for _, v in ipairs(workspace:GetDescendants()) do
            if v.Name == (espName .. "_HL") or v.Name == (espName .. "_BG") or v.Name == (espName .. "_BoxBG") then
                v:Destroy()
            end
            if v:IsA("BasePart") then
                local origTrans = v:GetAttribute("OrigTrans_" .. espName)
                if origTrans ~= nil then
                    v.Transparency = origTrans
                    v:SetAttribute("OrigTrans_" .. espName, nil)
                end
            end
        end
    end)
end

task.spawn(function()
    local function GetSlotText(entity)
        local slotsFolder = entity:FindFirstChild("Slots", true) or entity:FindFirstChild("slots", true)
        if not slotsFolder then
            local plr = game.Players:GetPlayerFromCharacter(entity)
            if plr then slotsFolder = plr:FindFirstChild("Slots", true) or plr:FindFirstChild("slots", true) end
        end
        if not slotsFolder then return nil end

        local slotsData = {}
        for i = 1, 10 do
            local s = slotsFolder:FindFirstChild("Slot" .. i) or slotsFolder:FindFirstChild("Slot " .. i)
            if s then
                local success, rawValue = pcall(function() return s.Value end)
                if success then
                    local val = "None"
                    if typeof(rawValue) == "Instance" then val = rawValue.Name
                    elseif rawValue ~= nil then val = tostring(rawValue) end
                    if val == "" or val == "nil" or val:match("^%s*$") then val = "None" end
                    table.insert(slotsData, "Slot " .. i .. " : " .. val)
                end
            end
        end
        if #slotsData == 0 then return nil end

        local line1, line2 = {}, {}
        for i, text in ipairs(slotsData) do
            if i <= 2 then table.insert(line1, text) else table.insert(line2, text) end
        end

        local finalText = table.concat(line1, "   ")
        if #line2 > 0 then finalText = finalText .. "\n" .. table.concat(line2, "   ") end
        local equippedTool = entity:FindFirstChildOfClass("Tool")
        if equippedTool then finalText = finalText .. "\n[ Holding : " .. equippedTool.Name .. " ]" end

        return finalText
    end

    local function UpdateSlotESP(target, text)
        if not target or not target.Parent then return end
        local root = target:FindFirstChild("HumanoidRootPart") or target.PrimaryPart or target:FindFirstChildWhichIsA("BasePart")
        if not root then return end

        local bg = root:FindFirstChild("ESP_Slot_BG") or Instance.new("BillboardGui")
        bg.Name = "ESP_Slot_BG"
        bg.Size = UDim2.new(0, 300, 0, 70) 
        bg.AlwaysOnTop = true
        bg.ExtentsOffsetWorldSpace = Vector3.new(0, -4.5, 0)

        local txt = bg:FindFirstChild("NameLabel") or Instance.new("TextLabel")
        txt.Name = "NameLabel"
        txt.Size = UDim2.new(1, 0, 1, 0)
        txt.BackgroundTransparency = 1
        txt.Text = text
        txt.TextColor3 = Options.SlotColor.Value
        txt.Font = Enum.Font.SourceSansBold
        txt.TextSize = 14
        txt.TextWrapped = true
        txt.TextStrokeTransparency = 0.6
        txt.AnchorPoint = Vector2.new(0.5, 0)
        txt.Position = UDim2.new(0.5, 0, 0, 0)
        txt.Parent = bg
        bg.Parent = root
    end

    while task.wait(1) do
        if Toggles.SlotEsp and Toggles.SlotEsp.Value then
            pcall(function()
                for _, plr in ipairs(game.Players:GetPlayers()) do
                    if plr.Character and plr ~= LocalPlayer then
                        if plr.Character:FindFirstChild("Weapon", true) then
                            local root = plr.Character:FindFirstChild("HumanoidRootPart") or plr.Character.PrimaryPart
                            if root and root:FindFirstChild("ESP_Slot_BG") then root.ESP_Slot_BG:Destroy() end
                        else
                            local text = GetSlotText(plr.Character)
                            if text then UpdateSlotESP(plr.Character, text) end
                        end
                    end
                end

                local playersFolder = GetTargetFolder("Players")
                if playersFolder then
                    for _, bot in ipairs(playersFolder:GetChildren()) do
                        if bot:IsA("Model") and bot ~= LocalPlayer.Character then
                            if bot:FindFirstChild("Weapon", true) then
                                local root = bot:FindFirstChild("HumanoidRootPart") or bot.PrimaryPart
                                if root and root:FindFirstChild("ESP_Slot_BG") then root.ESP_Slot_BG:Destroy() end
                            else
                                local text = GetSlotText(bot)
                                if text then UpdateSlotESP(bot, text) end
                            end
                        end
                    end
                end
            end)
        end
    end
end)

local ReplicatedStorage = game:GetService("ReplicatedStorage")

local function getRemotes()
    local events = ReplicatedStorage:FindFirstChild("Events")
    if events then return events:FindFirstChild("PlayEmote"), events:FindFirstChild("ModelAnimation") end
    return nil, nil
end

local function stopEmote()
    local char = LocalPlayer.Character
    if currentTrack then currentTrack:Stop() currentTrack = nil end
    local _, modelAnimRemote = getRemotes()
    if activeObject and char and modelAnimRemote then
        modelAnimRemote:FireServer(1, activeObject.uuid, activeObject.part, char)
        activeObject = nil
    end
end

local function playEmote()
    stopEmote()
    local emoteName = selectedType .. "Emote" .. selectedNum
    local char = LocalPlayer.Character
    if not char then return end

    local playEmoteRemote, modelAnimRemote = getRemotes()
    if playEmoteRemote then playEmoteRemote:FireServer(emoteName) end

    local objData = EmoteObjects[emoteName]
    if objData and modelAnimRemote then
        modelAnimRemote:FireServer(0, objData.uuid, objData.part, char)
        activeObject = objData 
    end

    local anim = nil
    for _, v in pairs(game:GetDescendants()) do
        if v.Name == emoteName and v:IsA("Animation") then anim = v break end
    end
    if anim then
        local hum = char:FindFirstChildOfClass("Humanoid")
        if hum then
            currentTrack = hum:LoadAnimation(anim)
            currentTrack.Looped = true
            currentTrack:Play()
        end
    end
end

local function fireObjectAction(actionType)
    local char = LocalPlayer.Character
    if not char then return end
    local _, modelAnimRemote = getRemotes()
    if not modelAnimRemote then return end
    local uuidsToFire = {}
    local partsToFire = {}

    if selectedCustomUUIDs["All"] then
        for _, uuid in pairs(ItemNameToUUID) do table.insert(uuidsToFire, uuid) end
    else
        for name, isSelected in pairs(selectedCustomUUIDs) do
            if isSelected and ItemNameToUUID[name] then table.insert(uuidsToFire, ItemNameToUUID[name]) end
        end
    end

    if selectedCustomParts["All"] then
        for _, part in ipairs(baseBodyPartsList) do table.insert(partsToFire, part) end
    else
        for partName, isSelected in pairs(selectedCustomParts) do
            if isSelected and partName ~= "All" then table.insert(partsToFire, partName) end
        end
    end

    if #uuidsToFire == 0 or #partsToFire == 0 then return end
    for _, uuid in ipairs(uuidsToFire) do
        for _, part in ipairs(partsToFire) do modelAnimRemote:FireServer(actionType, uuid, part, char) end
    end
end

    Func = function()
        local Player = game:GetService("Players").LocalPlayer
        local Character = Player.Character
        local Root = Character and Character:FindFirstChild("HumanoidRootPart")
        local Hum = Character and Character:FindFirstChildOfClass("Humanoid")
        local Bot = workspace:FindFirstChild("BOT") 
        if not (Bot and Bot:IsA("Model")) then
            for _, v in pairs(workspace:GetDescendants()) do
                if v.Name == "BOT" and v:IsA("Model") then Bot = v break end
            end
        end

        if not Bot or not Root or not Hum then
            Library:Notify("Error: Not Found Bot", 3)
            return 
        end

        local BotHum = Bot:FindFirstChildOfClass("Humanoid")
        local LowerTorso = Bot:FindFirstChild("LowerTorso")
        local UpperTorso = Bot:FindFirstChild("UpperTorso")
        local FallbackPart = Bot:FindFirstChild("HumanoidRootPart")

        task.spawn(function()
            local OldPos = Root.CFrame
            local OldFPDH = workspace.FallenPartsDestroyHeight
            workspace.FallenPartsDestroyHeight = -50000 
            if BotHum then workspace.CurrentCamera.CameraSubject = BotHum elseif LowerTorso then workspace.CurrentCamera.CameraSubject = LowerTorso end

            Hum.PlatformStand = true
            local BV = Instance.new("BodyVelocity")
            BV.Parent = Root
            BV.MaxForce = Vector3.new(math.huge, math.huge, math.huge)
            BV.Velocity = Vector3.new(0, 0, 0) 

            local StartTime = tick()
            local Connection
            local HasFlung = false

            Connection = game:GetService("RunService").Heartbeat:Connect(function()
                if tick() - StartTime > 3 or not Bot.Parent or not Character.Parent or HasFlung then
                    if Connection then Connection:Disconnect() end
                    return
                end

                local targetList = {}
                if LowerTorso then table.insert(targetList, LowerTorso) end
                if UpperTorso then table.insert(targetList, UpperTorso) end
                local TargetPart = (#targetList > 0) and targetList[math.random(1, #targetList)] or FallbackPart 
                if not TargetPart then return end

                if TargetPart.AssemblyLinearVelocity.Magnitude > 200 then
                    HasFlung = true
                    return
                end

                for _, part in pairs(Character:GetDescendants()) do
                    if part:IsA("BasePart") then part.CanCollide = false end
                end
                local ChaosRot = CFrame.Angles(math.rad(math.random(-36000, 36000)), math.rad(math.random(-36000, 36000)), math.rad(math.random(-36000, 36000)))
                local FrontOffset = CFrame.new(math.random(-5, 5) / 10, math.random(-5, 5) / 10, 5)

                Root.CFrame = TargetPart.CFrame * FrontOffset * ChaosRot
                Root.AssemblyLinearVelocity = Vector3.new(9e18, 9e18, 9e18)
                Root.AssemblyAngularVelocity = Vector3.new(9e18, 9e18, 9e18)
            end)

            task.wait(3)
            if Connection then Connection:Disconnect() end
            if BV then BV:Destroy() end
            Hum.PlatformStand = false
            workspace.CurrentCamera.CameraSubject = Hum
            workspace.FallenPartsDestroyHeight = OldFPDH
            Root.CFrame = OldPos
            Root.AssemblyLinearVelocity = Vector3.new(0, 0, 0)
            Root.AssemblyAngularVelocity = Vector3.new(0, 0, 0)
            local RS = game:GetService("ReplicatedStorage")
            local Ev = RS:FindFirstChild("Events")
            if Ev then
                if Ev:FindFirstChild("Leave") then Ev.Leave:FireServer() end
                task.wait(0.1)
                if Ev:FindFirstChild("Join") then Ev.Join:FireServer() end
            end
        end)
    end
})

    ToolList = {}
    ToolNames = {}
    local nameCount = {}
    local tf = workspace:FindFirstChild("Tools", true) or workspace:FindFirstChild("Items", true)
    if tf then
        for _, v in pairs(tf:GetChildren()) do
            if v:IsA("Model") or v:IsA("Tool") or v:IsA("BasePart") then
                local oName = v.Name
                nameCount[oName] = (nameCount[oName] or 0) + 1
                local dName = oName .. " [" .. tostring(nameCount[oName]) .. "]"
                table.insert(ToolNames, dName)
                ToolList[dName] = v
            end
        end
    end
    table.sort(ToolNames)
    Options.ToolDropdown:SetValues(#ToolNames > 0 and ToolNames or {"No Items Found"})
end })

        if hrp and pos then hrp.CFrame = pos.CFrame + Vector3.new(0, 3, 0) end
    end
end})

task.spawn(function()
    while task.wait(0.5) do
        if Library.Unloaded then
            if attackGui then attackGui:Destroy() end
            _G.PlayerEspEnabled = false
            _G.ItemEspEnabled = false
            _G.TrapEspEnabled = false
            if _G.HiddenCenterParts and #_G.HiddenCenterParts > 0 then
                for _, data in pairs(_G.HiddenCenterParts) do
                    if data.part and data.parent then data.part.Parent = data.parent end
                end
                _G.HiddenCenterParts = {}
            end
            break
        end
    end
end)
